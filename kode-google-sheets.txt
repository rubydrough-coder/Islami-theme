const SHEET_ID='1Q_tFVNaNsM1Q-w5tZF3QMN6Q0rBnu-Xbr01YJM1pLUM';

function safe(v){v=String(v);return /^[=+\-@]/.test(v)?"'"+v:v}

function doGet(){
  const s=SpreadsheetApp.openById(SHEET_ID).getSheets()[0];
  const rows=s.getDataRange().getValues().slice(1).reverse().slice(0,200);
  const out=rows.map(x=>({n:x[1],h:x[2],m:x[3]}));
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e){
  const d=JSON.parse(e.postData.contents);
  const s=SpreadsheetApp.openById(SHEET_ID).getSheets()[0];
  if(s.getLastRow()===0)s.appendRow(['Waktu','Nama','Hadir','Ucapan']);
  s.appendRow([new Date(),safe(String(d.n).slice(0,40)),safe(String(d.h).slice(0,20)),safe(String(d.m).slice(0,300))]);
  return ContentService.createTextOutput('ok');
}
